----------------------------------------------------------------
 PROJECT OVERVIEW
----------------------------------------------------------------

This is a complete, real-world Terraform project that:
  1. Creates an AWS VPC, subnet, and security group
  2. Launches an EC2 instance (server)
  3. Installs Docker on it via user_data
  4. Pulls and runs the log-generator container (from your HSBC project)
  5. Uses variables, locals, outputs, modules, and remote state
  6. Follows all best practices

This mirrors your actual log-generator-pipeline project at HSBC.

 FILE: providers.tf

  terraform {
    required_version = ">= 1.5, < 2.0"

    required_providers {
      aws = {
        source  = "hashicorp/aws"
        version = "~> 5.0"
      }
    }

    backend "s3" {
      bucket         = "hsbc-terraform-state"
      key            = "log-generator/terraform.tfstate"
      region         = "us-east-1"
      encrypt        = true
      dynamodb_table = "terraform-state-lock"
    }
  }

  provider "aws" {
    region = var.aws_region

    default_tags {
      tags = {
        ManagedBy   = "Terraform"
        Project     = "log-generator"
        Team        = "PDPOnPrem"
        Owner       = "Chandu"
      }
    }
  }

 FILE: locals.tf
----------------------------------------------------------------

  locals {
    # Naming prefix used across all resources
    prefix = "${var.project}-${var.environment}"
    # = "log-generator-dev" or "log-generator-prod"

    # Common tags applied to all resources
    common_tags = {
      Environment = var.environment
      Project     = var.project
    }

    # Environment-based instance type (override if not set in tfvars)
    resolved_instance_type = var.instance_type != "" ? var.instance_type : (
      var.environment == "prod" ? "t2.large" :
      var.environment == "uat"  ? "t2.small" :
      "t2.micro"
    )
  }

----------------------------------------------------------------
 FILE: main.tf
----------------------------------------------------------------

  # ----- NETWORKING MODULE -----
  module "networking" {
    source = "./modules/networking"

    prefix      = local.prefix
    vpc_cidr    = var.vpc_cidr
    environment = var.environment
    common_tags = local.common_tags
  }

  # ----- COMPUTE MODULE -----
  module "compute" {
    source = "./modules/compute"

    prefix          = local.prefix
    environment     = var.environment
    instance_type   = local.resolved_instance_type
    key_pair_name   = var.key_pair_name
    subnet_id       = module.networking.public_subnet_id
    security_group_id = module.networking.security_group_id
    docker_image    = var.docker_image
    nexus_username  = var.nexus_username
    nexus_password  = var.nexus_password
    common_tags     = local.common_tags
  }

----------------------------------------------------------------
 FILE: modules/networking/main.tf
----------------------------------------------------------------

  resource "aws_vpc" "main" {
    cidr_block           = var.vpc_cidr
    enable_dns_hostnames = true
    enable_dns_support   = true

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-vpc"
    })
  }

  resource "aws_internet_gateway" "main" {
    vpc_id = aws_vpc.main.id

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-igw"
    })
  }

  resource "aws_subnet" "public" {
    vpc_id                  = aws_vpc.main.id
    cidr_block              = cidrsubnet(var.vpc_cidr, 8, 1)    # 10.0.1.0/24
    map_public_ip_on_launch = true
    availability_zone       = data.aws_availability_zones.available.names[0]

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-public-subnet"
      Type = "Public"
    })
  }

  resource "aws_route_table" "public" {
    vpc_id = aws_vpc.main.id

    route {
      cidr_block = "0.0.0.0/0"
      gateway_id = aws_internet_gateway.main.id
    }

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-public-rt"
    })
  }

  resource "aws_route_table_association" "public" {
    subnet_id      = aws_subnet.public.id
    route_table_id = aws_route_table.public.id
  }

  resource "aws_security_group" "server_sg" {
    name        = "${var.prefix}-server-sg"
    description = "Security group for ${var.prefix} server"
    vpc_id      = aws_vpc.main.id

    ingress {
      description = "SSH"
      from_port   = 22
      to_port     = 22
      protocol    = "tcp"
      cidr_blocks = ["10.0.0.0/8"]    # Internal HSBC range only!
    }

    egress {
      description = "All outbound"
      from_port   = 0
      to_port     = 0
      protocol    = "-1"
      cidr_blocks = ["0.0.0.0/0"]
    }

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-server-sg"
    })
  }

  data "aws_availability_zones" "available" {
    state = "available"
  }

 FILE: modules/compute/main.tf
----------------------------------------------------------------

  data "aws_ami" "ubuntu" {
    most_recent = true
    owners      = ["099720109477"]    # Canonical

    filter {
      name   = "name"
      values = ["ubuntu/images/hvm-ssd/ubuntu-jammy-22.04-amd64-server-*"]
    }
  }

  resource "aws_instance" "server" {
    ami                    = data.aws_ami.ubuntu.id
    instance_type          = var.instance_type
    key_name               = var.key_pair_name
    subnet_id              = var.subnet_id
    vpc_security_group_ids = [var.security_group_id]

    user_data = <<-EOF
      #!/bin/bash
      set -e

      # Update and install Docker
      apt-get update -y
      apt-get install -y docker.io
      systemctl start docker
      systemctl enable docker

      # Log in to Nexus registry
      echo "${var.nexus_password}" | \
        docker login nexus3.systems.uk.hsbc:8083 \
          -u "${var.nexus_username}" \
          --password-stdin

      # Pull the log-generator image
      docker pull ${var.docker_image}

      # Run the log-generator container
      docker run -d \
        --name log-generator \
        --restart unless-stopped \
        ${var.docker_image}

      echo "Setup complete!"
    EOF

    tags = merge(var.common_tags, {
      Name = "${var.prefix}-server"
    })
  }

----------------------------------------------------------------
 DEPLOYMENT STEPS (Full Workflow)
----------------------------------------------------------------

  # 1. Set credentials (never in code!)
  export AWS_ACCESS_KEY_ID="..."
  export AWS_SECRET_ACCESS_KEY="..."
  export TF_VAR_nexus_username="your-nexus-user"
  export TF_VAR_nexus_password="your-nexus-password"

  # 2. Initialize
  terraform init

  # 3. Validate
  terraform validate
  terraform fmt -check

  # 4. Plan for dev
  terraform plan -var-file="dev.tfvars" -out=dev.plan

  # 5. Review the plan carefully!

  # 6. Apply
  terraform apply dev.plan

  # 7. Check outputs
  terraform output

  # 8. SSH and verify
  ssh -i ~/.ssh/hsbc-dev-keypair.pem ubuntu@$(terraform output -raw server_public_ip)
  docker logs log-generator

  # 9. Clean up (dev only!)
  terraform destroy -var-file="dev.tfvars"

  ✓ merge() function for tag composition
  ✓ Outputs including useful commands
  ✓ .gitignore for secrets
  ✓ Separate tfvars per environment
  ✓ Tagging strategy
  ✓ Security group with minimal access
  ✓ Naming convention: {project}-{environment}-{resource}
