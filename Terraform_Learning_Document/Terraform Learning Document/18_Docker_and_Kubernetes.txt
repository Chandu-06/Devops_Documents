----------------------------------------------------------------
 WHY MANAGE DOCKER/KUBERNETES WITH TERRAFORM?
----------------------------------------------------------------

You already know Docker and Kubernetes from your onboarding at HSBC.
Terraform can manage them too!

Instead of running "docker run" or "kubectl apply" manually,
you can define Docker containers and Kubernetes resources as
Terraform code — giving you:
  - Version control for your container/K8s configs
  - Consistent deployments across environments
  - Integrated lifecycle management (create/update/destroy)
  - Dependency management between infra and app deployments

----------------------------------------------------------------
 PART 1: TERRAFORM WITH DOCKER
----------------------------------------------------------------

The Docker provider lets Terraform manage Docker images and containers.

First, install the Docker provider:

  terraform {
    required_providers {
      docker = {
        source  = "kreuzwerker/docker"
        version = "~> 3.0"
      }
    }
  }

  provider "docker" {
    host = "unix:///var/run/docker.sock"    # Local Docker socket
    # Or for remote Docker:
    # host = "tcp://remote-docker-host:2376"
  }

--- Pull an Image ---

  resource "docker_image" "nginx" {
    name         = "nginx:latest"
    keep_locally = true    # Don't delete image when resource is destroyed
  }

  resource "docker_image" "log_generator" {
    name         = "nexus3.systems.uk.hsbc:8083/hsbc-242921-pdponprem/dev/log-generator:latest"
    keep_locally = true
  }

--- Run a Container ---

  resource "docker_container" "nginx_server" {
    name  = "my-nginx"
    image = docker_image.nginx.image_id    # Reference the pulled image

    ports {
      internal = 80      # Port inside container
      external = 8080    # Port on your host
    }

    restart = "always"   # Restart policy

    env = [
      "APP_ENV=dev",
      "LOG_LEVEL=info"
    ]

    volumes {
      container_path = "/usr/share/nginx/html"
      host_path      = "/home/chandu/html"
      read_only      = true
    }
  }

--- Output container IP ---

  output "container_ip" {
    value = docker_container.nginx_server.network_data[0].ip_address
  }

--- Docker Network ---

  resource "docker_network" "app_network" {
    name   = "hsbc-app-network"
    driver = "bridge"
  }

  resource "docker_container" "app" {
    name    = "my-app"
    image   = docker_image.log_generator.image_id
    networks_advanced {
      name = docker_network.app_network.name
    }
  }

--- Docker Volume ---

  resource "docker_volume" "app_data" {
    name = "app-data-volume"
  }

  resource "docker_container" "app" {
    name  = "my-app"
    image = docker_image.log_generator.image_id

    volumes {
      volume_name    = docker_volume.app_data.name
      container_path = "/app/data"
    }
  }

--- Full Docker Example: log-generator (HSBC Context) ---

  terraform {
    required_providers {
      docker = {
        source  = "kreuzwerker/docker"
        version = "~> 3.0"
      }
    }
  }

  provider "docker" {
    host = "unix:///var/run/docker.sock"
  }

  resource "docker_image" "log_generator" {
    name         = "nexus3.systems.uk.hsbc:8083/hsbc-242921-pdponprem/dev/log-generator:latest"
    keep_locally = true
  }

  resource "docker_container" "log_generator" {
    name    = "log-generator-tf"
    image   = docker_image.log_generator.image_id
    restart = "on-failure"

    env = [
      "LOG_INTERVAL=5",
      "APP_ENV=dev"
    ]
  }

  output "container_id" {
    value = docker_container.log_generator.id
  }

 PART 2: TERRAFORM WITH KUBERNETES
----------------------------------------------------------------

The Kubernetes provider lets Terraform manage Kubernetes resources:
Deployments, Services, ConfigMaps, Namespaces, etc.

Install the Kubernetes provider:

  terraform {
    required_providers {
      kubernetes = {
        source  = "hashicorp/kubernetes"
        version = "~> 2.0"
      }
    }
  }

  provider "kubernetes" {
    config_path    = "~/.kube/config"     # Use local kubeconfig
    config_context = "my-cluster"         # Which context to use

    # Or for in-cluster (running inside Kubernetes):
    # host                   = var.k8s_host
    # cluster_ca_certificate = base64decode(var.cluster_ca)
    # token                  = var.cluster_token
  }

--- Namespace ---

  resource "kubernetes_namespace" "dev" {
    metadata {
      name = "dev"
      labels = {
        environment = "dev"
        managed-by  = "terraform"
      }
    }
  }

--- Deployment ---

  resource "kubernetes_deployment" "log_generator" {
    metadata {
      name      = "log-generator"
      namespace = kubernetes_namespace.dev.metadata[0].name
      labels = {
        app = "log-generator"
      }
    }

    spec {
      replicas = 2    # Number of pods

      selector {
        match_labels = {
          app = "log-generator"
        }
      }

      template {
        metadata {
          labels = {
            app = "log-generator"
          }
        }

        spec {
          container {
            name  = "log-generator"
            image = "nexus3.systems.uk.hsbc:8083/hsbc-242921-pdponprem/dev/log-generator:latest"

            resources {
              limits = {
                cpu    = "500m"
                memory = "128Mi"
              }
              requests = {
                cpu    = "250m"
                memory = "64Mi"
              }
            }

            env {
              name  = "APP_ENV"
              value = "dev"
            }

            env {
              name  = "LOG_INTERVAL"
              value = "5"
            }
          }

          image_pull_secrets {
            name = "nexus-registry-secret"    # Secret for private registry
          }
        }
      }
    }
  }

--- Service ---

  resource "kubernetes_service" "log_generator" {
    metadata {
      name      = "log-generator-svc"
      namespace = kubernetes_namespace.dev.metadata[0].name
    }

    spec {
      selector = {
        app = "log-generator"    # Matches pods with this label
      }

      port {
        port        = 80
        target_port = 8080
      }

      type = "ClusterIP"    # Internal only (ClusterIP / NodePort / LoadBalancer)
    }
  }

--- ConfigMap ---

  resource "kubernetes_config_map" "app_config" {
    metadata {
      name      = "log-generator-config"
      namespace = kubernetes_namespace.dev.metadata[0].name
    }

    data = {
      LOG_INTERVAL = "5"
      APP_ENV      = "dev"
      LOG_LEVEL    = "INFO"
    }
  }

  # Use the ConfigMap in the Deployment:
  # env_from {
  #   config_map_ref {
  #     name = kubernetes_config_map.app_config.metadata[0].name
  #   }
  # }

--- Secret ---

  resource "kubernetes_secret" "registry_credentials" {
    metadata {
      name      = "nexus-registry-secret"
      namespace = kubernetes_namespace.dev.metadata[0].name
    }

    type = "kubernetes.io/dockerconfigjson"

    data = {
      ".dockerconfigjson" = jsonencode({
        auths = {
          "nexus3.systems.uk.hsbc:8083" = {
            username = var.nexus_username
            password = var.nexus_password
            auth     = base64encode("${var.nexus_username}:${var.nexus_password}")
          }
        }
      })
    }
  }

--- Outputs ---

  output "deployment_name" {
    value = kubernetes_deployment.log_generator.metadata[0].name
  }

  output "namespace" {
    value = kubernetes_namespace.dev.metadata[0].name
  }

  output "service_cluster_ip" {
    value = kubernetes_service.log_generator.spec[0].cluster_ip
  }

 COMBINING AWS + KUBERNETES (Real-World Pattern)
----------------------------------------------------------------

A common pattern: Terraform creates an EKS cluster (AWS K8s),
then deploys apps to it in the same run.

  # Step 1: Create EKS cluster
  module "eks" {
    source          = "terraform-aws-modules/eks/aws"
    cluster_name    = "hsbc-payments-cluster"
    cluster_version = "1.28"
    vpc_id          = module.vpc.vpc_id
    subnet_ids      = module.vpc.private_subnets
  }

  # Step 2: Configure K8s provider to use the new EKS cluster
  provider "kubernetes" {
    host                   = module.eks.cluster_endpoint
    cluster_ca_certificate = base64decode(module.eks.cluster_certificate_authority_data)
    token                  = data.aws_eks_cluster_auth.cluster.token
  }

  # Step 3: Deploy application to the cluster
  resource "kubernetes_deployment" "app" {
    # ... deployment config ...
  }

----------------------------------------------------------------
 SUMMARY
----------------------------------------------------------------

  DOCKER PROVIDER:
    - Manage images, containers, networks, volumes
    - provider "docker" connects via Docker socket
    - docker_image → pull images
    - docker_container → run containers

  KUBERNETES PROVIDER:
    - Manage Namespaces, Deployments, Services, ConfigMaps, Secrets
    - provider "kubernetes" uses kubeconfig or direct credentials
    - kubernetes_namespace, kubernetes_deployment, kubernetes_service
    - Replaces manual kubectl apply with code-managed resources

  KEY BENEFITS:
    - Same Terraform workflow (init, plan, apply, destroy)
    - Infrastructure AND application in the same codebase
    - Consistent environments (dev/uat/prod) from same code
    - Integrated dependency management
